//java
class Solution {
    public void nextPermutation(int[] nums) {
        int n = nums.length;

        int i = n - 2;

        while (i >= 0 && nums[i] >= nums[i + 1]) {
            i--;
        }

        if (i >= 0) {
            int j = n - 1;

            while (nums[j] <= nums[i]) {
                j--;
            }


            int temp = nums[i];
            nums[i] = nums[j];
            nums[j] = temp;
        }

        int left = i + 1;
        int right = n - 1;

        while (left < right) {
            int temp = nums[left];
            nums[left] = nums[right];
            nums[right] = temp;

            left++;
            right--;
        }
    }
}
class Main {
    public static void main(String[] args) {
        int[] data={1, 2, 3, 6, 5,4};
        Solution obj=new Solution();
        obj.nextPermutation(data);
        for(int i=0;i<data.length;i++){
        System.out.print(data[i]+" ");}
    }
}

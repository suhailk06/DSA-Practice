//java

class Main {
    public static void main(String[] args) {
        int[] data={1, 4, 3, 2, 6, 5};
        int n=data.length;
        for (int i = 0; i < n/2; i++) {
            int temp=data[i];
            data[i]=data[n-1-i];
            data[n-i-1]=temp;
            
        }
        System.out.println("reverse data");
        for(int i=0;i<n;i++){
            System.out.print(data[i]+" ");
            }
        
    }
}

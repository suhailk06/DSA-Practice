//java
import java.util.*;
class Main {
    public static void main(String[] args) {
        int[] data={3 ,1, 2 ,5 ,3};
        Arrays.sort(data);
        HashMap<Integer,Integer> key=new HashMap<>();
        for(int i=1;i<data.length;i++){
            key.put(data[i]-data[i-1],key.getOrDefault(data[i]-data[i-1], 1) + 1);
        }
        int max=0;
        int k=0;
        for(var x:key.keySet()){
            if(max<key.get(x)){
                k=x;
                max=key.get(x);
            }
        }
        for(int i=0;i<data.length;i++){
            if(data[i]!=k+i){
                System.out.println(data[i]+" to "+(k+i));
            }
        }
            
    }



//python
data = [3, 1, 2, 5, 3]
data.sort() 

diff_count = {}
for i in range(1, len(data)):
    diff = data[i] - data[i-1]
    diff_count[diff] = diff_count.get(diff, 1) + 1
max_count = 0
common_diff = 0
for diff in diff_count:
    if max_count < diff_count[diff]:
        common_diff = diff
        max_count = diff_count[diff]
for i in range(len(data)):
    if data[i] != common_diff + i:
        print(f"{data[i]} to {common_diff + i}")
        data[i] = common_diff + i

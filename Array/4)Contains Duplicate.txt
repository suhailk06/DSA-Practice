//c
#include <stdio.h>

int main() {
    int data[]= {1,2,3,1};
    int l=sizeof(data)/sizeof(data[0]);
    int flag=0;
    
    for(int i=0;i<l && !flag;i++){ 
        for(int j=0;j<l;j++){
            if(i!=j && data[i]==data[j]){
                flag=1;
                break;  
            }
        }
    }
    
    flag==1?printf("true"):printf("false");
    return 0;
}

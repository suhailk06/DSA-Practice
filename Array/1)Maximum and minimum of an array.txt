//java
import java.util.*;
class Main {
    public static void main(String[] args) {
        int[] data={22, 14, 8, 17, 35, 3};
        int min = data[0];
        int max = data[0];

        for (int i = 1; i < data.length; i++) {
            if (data[i] < min)
                min = data[i];

            if (data[i] > max)
            max = data[i];
        }

System.out.println("Min: " + min);
System.out.println("Max: " + max);
        
    }
}

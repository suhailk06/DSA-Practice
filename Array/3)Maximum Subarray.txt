//java
class Main {
    public static void main(String[] args) {
        int[] data={-2,1,-3,4,-1,2,1,-5,4};
        int current=data[0];
        int max=data[0];
        for(int i=1;i<data.length;i++){
            current=Math.max(data[i],data[i]+current);
            max=Math.max(max,current);

        }
        System.out.println("max:"+max);
    }
}

//c
#include <stdio.h>

int main() {
    int data[]={-2,1,-3,4,-1,2,1,-5,4};
    int l=sizeof(data) / sizeof(data[0]);
    int current=data[0];
    int maxval=data[0];
    for(int i=1;i<l;i++){
        current = (current + data[i] > data[i]) ? current + data[i] : data[i];
        if (maxval<current){
            maxval=current;
        }
    }
    printf("max is:%d",maxval);
    return 0;
}

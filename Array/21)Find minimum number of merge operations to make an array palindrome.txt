//java
import java.util.*;
class Solve{
    public int solve(int[] arr)
    {
        int n=arr.length;
        int ans = 0;
        for (int i=0,j=n-1; i<=j;)
        {
            if (arr[i] == arr[j])
            {
                i++;
                j--;
            }

            else if (arr[i] > arr[j])
            {
                j--;
                arr[j] += arr[j+1] ;
                ans++;
            }
            else
            {
                i++;
                arr[i] += arr[i-1];
                ans++;
            }
        }

        return ans;
    }
}
class Main {
    public static void main(String[] args) {
        int[] arr = {11, 14, 15, 99};
        Solve obj=new Solve();
        System.out.println(obj.solve(arr));
    }
}

//java
class Solution {
    public int[] productExceptSelf(int[] nums) {
        int[] ans=new int[nums.length];
        int[] ans2=new int[nums.length];
        ans[0]=1;
        ans2[nums.length-1]=1;
        for(int i=1;i<ans.length;i++){
            ans[i]=ans[i-1]*nums[i-1];
        }
        for(int i=ans2.length-2;i>=0;i--){
            ans2[i]=ans2[i+1]*nums[i+1];
        }
        for(int i=0;i<nums.length;i++){
            nums[i]=ans[i]*ans2[i];
        }
        return nums;
    }
}
class Main {
    public static void main(String[] args) {
      int[] data={1,2,3,4};
      Solution obj=new Solution();
      data=obj.productExceptSelf(data);
      for(int i=0;i<data.length;i++){
        System.out.println(data[i]);
    }}
}

def productExceptSelf(nums):
    ans = []
    ans2 = []

    ans.append(1)
    ans2.append(1)

    n = len(nums) - 1

    for i in range(1, len(nums)):
        ans.append(ans[i - 1] * nums[i - 1])
        ans2.append(ans2[i - 1] * nums[n - i + 1])

    for i in range(len(nums)):
        ans[i] = ans[i] * ans2[n - i]

    return ans
data=[1,2,3,4]
print(productExceptSelf(data))

class Solution {
    public String largestNumber(int[] nums) {
        String[] list = new String[nums.length];

        for (int i = 0; i < nums.length; i++) {
            list[i] = nums[i] + "";
        }

        Arrays.sort(list, (a, b) -> (b + a).compareTo(a + b));

        if (list[0].equals("0")) {
            return "0";
        }

        String ans = "";

        for (int i = 0; i < list.length; i++) {
            ans += list[i];
        }

        return ans;
    }
}
